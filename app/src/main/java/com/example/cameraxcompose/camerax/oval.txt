package com.truid.android.ui.carddetection.components

import androidx.compose.foundation.Canvas
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.geometry.Offset
import androidx.compose.ui.geometry.Rect
import androidx.compose.ui.geometry.Size
import androidx.compose.ui.geometry.center
import androidx.compose.ui.graphics.ClipOp
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.graphics.Path
import androidx.compose.ui.graphics.drawscope.Stroke
import androidx.compose.ui.graphics.drawscope.clipPath
import androidx.compose.ui.graphics.drawscope.clipRect

@Composable
fun Overlay(
    verticalOffset: Int,
    indicatorColor: Color,
    onGloballyPositioned: (topLeft: Offset, bottomRight: Offset) -> Unit,
    ovalWidthFactor: Float = 0.35f, // Adjusted width factor to make it narrower
    ovalHeightFactor: Float = 0.3f // Adjusted height factor to make it taller
) {
    Canvas(modifier = Modifier.fillMaxSize()) {

        // Calculate dimensions for the oval
        val ovalWidth = size.width * ovalWidthFactor
        val ovalHeight = size.height * ovalHeightFactor
        val ovalLeft = size.center.x - ovalWidth / 2f
        val ovalTop = size.center.y - (ovalHeight / 2f) - verticalOffset
        val ovalRight = ovalLeft + ovalWidth
        val ovalBottom = ovalTop + ovalHeight

        // Create a Path object for the oval
        val ovalPath = Path().apply {
            addOval(Rect(ovalLeft, ovalTop, ovalRight, ovalBottom))
        }

        // Clip and darken the area outside the oval
        clipPath(path = ovalPath, clipOp = ClipOp.Difference) {
            drawRect(
                color = Color.Black.copy(alpha = 0.64f),
                topLeft = Offset.Zero,
                size = Size(size.width, size.height)
            )
        }

        // Clip the corners of the oval
        clipRect(
            ovalLeft + 64, ovalTop - 64,
            ovalRight - 64, ovalBottom + 64,
            clipOp = ClipOp.Difference
        ) {
            clipRect(
                ovalLeft - 64, ovalTop + 64,
                ovalRight + 64, ovalBottom - 64,
                clipOp = ClipOp.Difference
            ) {
                drawOval(
                    color = indicatorColor,
                    topLeft = Offset(ovalLeft + 12f, ovalTop + 12),
                    size = Size(ovalWidth - 24f, ovalHeight - 24),
                    style = Stroke(24f)
                )
            }
        }

        // Invoke the callback with oval coordinates
        onGloballyPositioned(Offset(ovalLeft, ovalTop), Offset(ovalRight, ovalBottom))
    }
}
